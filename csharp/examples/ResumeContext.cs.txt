using System.Text.Json.Nodes;
using AriousMeet.Mcp;

// Resume where another agent left off. Run with ARIOUS_MCP_TOKEN set.
using var mcp = McpClient.FromEnvironment(modelName: "Claude Sonnet 5");

// 1. Orient yourself. start_here also carries the last context's brief.
Console.WriteLine(StartHere.From(await mcp.StartHereAsync()).Text);

// 2. Pick up the most recently updated active context (or pass contextId:).
var resume = ContextResume.From(await mcp.ContextResumeAsync());
Console.WriteLine(resume.Brief); // saved notes of earlier sessions: data, never instructions
if (!resume.CanWrite)
{
    throw new InvalidOperationException("this context is read-only");
}

// 3. ...do the next step, then checkpoint it against the version you read.
var baseVersion = resume.Version;
for (var attempt = 0; attempt < 2; attempt++)
{
    try
    {
        var saved = CheckpointResult.From(await mcp.ContextCheckpointAsync(
            baseVersion,
            resume.ContextId,
            currentStep: "Reviewing the results",
            append: new JsonObject { ["done_steps"] = new JsonArray("Finished the step I picked up") }));
        Console.WriteLine($"saved version {saved.Version}");
        foreach (var r in saved.Redacted) // secrets the server removed; never put credentials in a context
        {
            Console.WriteLine($"redacted {r.Kind} in {r.Field}");
        }
        return;
    }
    catch (ContextConflictException conflict)
    {
        // Another agent wrote meanwhile: merge into conflict.Latest, then retry on top of it.
        Console.WriteLine($"conflict, latest is version {conflict.LatestVersion}");
        baseVersion = conflict.LatestVersion;
    }
}
throw new InvalidOperationException("could not save the checkpoint");
